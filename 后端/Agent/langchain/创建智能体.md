```python
from functools import lru_cache
from typing import Any, Literal

from langchain.agents import AgentState, create_agent
from langchain.agents.middleware.types import InputAgentState, OutputAgentState
from langchain_deepseek import ChatDeepSeek
from langgraph.graph.state import CompiledStateGraph

from app.config import AgentConfig
from app.modules.agents import deepseek_search_tool


@lru_cache
def get_ds_model(*, effort: Literal["none", "low", "high", "max"] = "high") -> ChatDeepSeek:
    """懒加载并缓存 DeepSeek 聊天模型实例。"""
    return ChatDeepSeek(
        model="deepseek-flash",
        api_key=AgentConfig["DEEPSEEK_API_KEY"],
        max_tokens=None,  # 不限制单次输出 token 数量
        max_retries=2,
        reasoning_effort=effort,
    )


def get_ds_model_with_effort(effort: Literal["none", "low", "high", "max"] = "high"):
    return get_ds_model().bind(reasoning_effort=effort)


@lru_cache
def get_agent() -> CompiledStateGraph[
    AgentState[Any], Any, InputAgentState, OutputAgentState[Any]
]:
    """懒加载并缓存 DeepSeek agent。输入输出都是带 messages 的 AgentState。"""
    return create_agent(
        model=get_ds_model(),
        tools=[deepseek_search_tool.deepseek_search],
        system_prompt="你是一个智能助手猫娘，非常可爱，需要时主动使用联网搜索工具。",
    )
```
