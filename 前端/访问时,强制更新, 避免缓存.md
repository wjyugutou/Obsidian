import type { Router } from 'vue-router'

const VERSION_KEY = 'app-update:entry'
const CHUNK_KEY = 'app-update:chunk'
const CHUNK_COOLDOWN_MS = 10_000

/**
 * 生产环境强制换到新包。
 * 入口 HTML 被浏览器或网关缓存时，旧页面会继续引用已经删掉的带哈希 chunk。
 * 打开页面时用 no-store 拉一份最新 index.html，入口脚本变了就整页替换到带时间戳的地址，绕开旧文档缓存。
 * 同一份新入口只跳一次；chunk 加载失败 10 秒内只跳一次，避免网关仍返回旧 HTML 时死循环。
 */
export function startAppUpdate(router: Router) {
  if (!import.meta.env.PROD)
    return

  stripReloadMark()

  let pendingHref = currentHref()

  router.beforeEach((to) => {
    pendingHref = withBase(to.fullPath)
  })

  window.addEventListener('vite:preloadError', (event) => {
    if (!markChunkReload())
      return
    event.preventDefault()
    replaceWithBust(pendingHref || currentHref())
  })

  void checkEntry()
}

function currentHref() {
  return `${location.pathname}${location.search}${location.hash}`
}

function withBase(fullPath: string) {
  const base = (import.meta.env.BASE_URL || '/').replace(/\/$/, '')
  const path = fullPath.startsWith('/') ? fullPath : `/${fullPath}`
  return `${base}${path}`
}

function stripReloadMark() {
  const url = new URL(location.href)
  if (!url.searchParams.has('__r'))
    return
  url.searchParams.delete('__r')
  history.replaceState(history.state, '', `${url.pathname}${url.search}${url.hash}`)
}

function replaceWithBust(href: string) {
  const url = new URL(href, location.origin)
  url.searchParams.set('__r', String(Date.now()))
  location.replace(url.href)
}

function currentModulePath() {
  const src = [...document.querySelectorAll('script[type="module"]')]
    .map(el => el.getAttribute('src') || '')
    .find(item => item.includes('/assets/'))
  return src ? normalizePath(src) : ''
}

function normalizePath(src: string) {
  try {
    return new URL(src, location.href).pathname
  }
  catch {
    return src
  }
}

function modulePathFromHtml(html: string) {
  const src = [...html.matchAll(/<script\b[^>]*>/gi)]
    .map(match => match[0])
    .filter(tag => /type=["']module["']/.test(tag))
    .map(tag => tag.match(/\bsrc=["']([^"']+)["']/i)?.[1] || '')
    .find(item => item.includes('/assets/'))
  return src ? normalizePath(src) : ''
}

async function fetchEntryPath() {
  const base = import.meta.env.BASE_URL || '/'
  const url = new URL('index.html', new URL(base, location.origin))
  url.searchParams.set('t', String(Date.now()))
  try {
    const res = await fetch(url.href, { cache: 'no-store' })
    if (!res.ok)
      return ''
    return modulePathFromHtml(await res.text())
  }
  catch {
    return ''
  }
}

async function checkEntry() {
  const local = currentModulePath()
  if (!local)
    return
  const remote = await fetchEntryPath()
  if (!remote || remote === local) {
    if (remote === local)
      removeStorage(VERSION_KEY)
    return
  }
  if (readStorage(VERSION_KEY) === remote)
    return
  writeStorage(VERSION_KEY, remote)
  replaceWithBust(currentHref())
}

function markChunkReload() {
  const at = Number(readStorage(CHUNK_KEY) || 0)
  if (Date.now() - at < CHUNK_COOLDOWN_MS)
    return false
  writeStorage(CHUNK_KEY, String(Date.now()))
  return true
}

function readStorage(key: string) {
  try {
    return sessionStorage.getItem(key) || ''
  }
  catch {
    return ''
  }
}

function writeStorage(key: string, value: string) {
  try {
    sessionStorage.setItem(key, value)
  }
  catch {
    // 隐私模式写不进 session 时仍继续跳转
  }
}

function removeStorage(key: string) {
  try {
    sessionStorage.removeItem(key)
  }
  catch {
    // 忽略
  }
}
